

# aga_tr_afr_swahili_stone_town (Stone Town Charter): coral-stone merchant houses on the Swahili coast. Lock: Swahili culture.
# #culture_group east_african_coast_group

    kept_at_conquest = no

    location_modifier = {
		local_maritime_presence = 0.1
		local_trades_per_burgher = 0.25
        local_trade_protection_factor = 0.05
		local_marketplace_building_levels = 5
		local_market_upgrade_building_levels = 5
        local_burghers_estate_power = 1.0
    }

    country_modifier = {
    }

    potential = {
        culture = {
            has_culture_group = culture_group:east_african_coast_group
        }
    }

    allow = {
        scope:target = {
			is_port = yes
			NOT = {
				has_town_rights = town_rights_type:staple_port
            }
        }
    }

    color = map_dene
}
